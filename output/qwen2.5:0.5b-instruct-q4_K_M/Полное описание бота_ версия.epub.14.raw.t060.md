    # Обработка сообщений из Telegram
    for update in dp.get_updates():

            user = update.message.from_user.username
            message = update.message.text

            if user not in db.get_users():  # Добавляем в базу данных
                db.add_new_user(user)
                
            elif user in db.get_users() and db.is_active(user) == 1:  # Доступный пользователь
                if db.is_blacklisted(message):  # Если сообщение было удалено
                    await message.reply("Вы уже заблокированы.")
                    break

                # Преобразуем текст сообщения в обычное сообщение для бота
                text = parse_message(message)
                
                # Обработка сообщений
                if text and user in db.get_users():

                        break

        break
