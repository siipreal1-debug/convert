    def __init__(self):
        self.vector_store = {}

    async def upsert_embedding(self, text: str, metadata: dict) -> str:
        """Store embedding with metadata for retrieval"""
        embedding_id = f"emb_{len(self.vector_store) + 1}"
        self.vector_store[embedding_id] = {
            "text": text,
            "metadata": metadata
        }
        return embedding_id

    async def retrieve_relevant(self, query: str, top_k: int = 3) -> List[dict]:
        """Retrieve top-k relevant embeddings based on query"""
        results = []
        for embed_id, data in self.vector_store.items():
            relevance = self._calculate_relevance(query, data["text"])
            if relevance > 0.3:  # Threshold for relevance
                results.append({
                    "id": embed_id,
                    "relevance": relevance,
                    "text": data["text"]
                })
        results.sort(key=lambda x: x["relevance"], reverse=True)
        return results[:top_k]

    def _calculate_relevance(self, query: str, text: str) -> float:
        # Simple relevance score (real implementation would use vector embeddings)
