```python
import tkinter as tk
import time
import os
import threading
from typing import Dict, Any, Optional

class UI:
    def __init__(self, db):
        """
        Initialize the UI component with database connection.
        
        Args:
            db: Database connection object
        """
        self.db = db
        self.root = tk.Tk()
        self.root.title("Bot System UI")
        self.root.geometry("800x600")
        
        # Create text area for logs
        self.text_area = tk.Text(self.root, height=20, width=50)
        self.text_area.pack(padx=10, pady=10)
        
        # Create buttons
        self.create_buttons()
        
        # Initialize UI state
        self.running = True
        self.paused = False
        self.clean_messages = False
        
        # Start UI thread
        self.thread = threading.Thread(target=self.run_ui)
        self.thread.daemon = True
        self.thread.start()
        
        # Add initial log message
        self.add_log("Bot system started. Ready for commands.")

    def create_buttons(self):
        """Create all UI buttons with proper functionality"""
        # Pause/Resume button
        self.pause_button = tk.Button(
            self.root, 
            text="Pause/Resume", 
            command=self.toggle_pause,
            width=20
        )
        self.pause_button.pack(pady=5)
        
        # Clean messages button
        self.clean_button = tk.Button(
            self.root, 
            text="Enable Clean Messages", 
            command=lambda: self.set_clean(True),
            width=25
        )
        self.clean_button.pack(pady=5)
        
        # Blacklist button
        self.blacklist_button = tk.Button(
            self.root, 
            text="Add to Blacklist", 
            command=lambda user="": self.add_blacklist(user),
            width=25
        )
        self.blacklist_button.pack(pady=5)
        
        # Reset sessions button
        self.reset_button = tk.Button(
            self.root, 
            text="Reset Sessions", 
            command=self.reset_sessions,
            width=20
        )
        self.reset_button.pack(pady=5)
        
        # Operator button
        self.operator_button = tk.Button(
            self.root, 
            text="Set Operator", 
            command=lambda: self.set_operator(""),
            width=20
        )
        self.operator_button.pack(pady=5)

    def run_ui(self):
        """Main UI loop that runs in a separate thread"""
        while self.running:
            try:
                # Check for pause state
                if self.db.paused:
                    time.sleep(1)
                else:
                    # Check queue and process items
                    try:
                        item = self.db.get_queue_item()
                        if item:
                            state = {
                                'user': item['user'],
                                'message': item['message'],
                                'chat_id': item['chat_id']
                            }
                            response = self.db.app.ainvoke(state)
                            
                            # Send response to Telegram
                            self.db.application.bot.send_message(
                                chat_id=item['chat_id'],
                                text=response
                            )
                            
                            # Update UI
                            self.add_response(item['user'], response)
                            self.db.remove_from_queue(item['id'])
                    except Exception as e:
                        self.log_error(e)
            except Exception as e:
                self.log_error(e)
                time.sleep(1)

    def add_log(self, message: str):
        """Add log message to UI"""
        self.text_area.insert(tk.END, f"Log: {message}\n")
        self.text_area.see(tk.END)  # Scroll to end

    def add_incoming(self, user: str, message: str):
        """Add incoming message to UI"""
        self.text_area.insert(tk.END, f"Incoming: {user} - {message}\n")
        self.text_area.see(tk.END)

    def add_response(self, user: str, response: str):
        """Add response to UI"""
        self.text_area.insert(tk.END, f"Response: {user} - {response}\n")
        self.text_area.see(tk.END)

    def log_error(self, e: Exception):
        """Log error to file and UI"""
        error_msg = f"Error: {str(e)}"
        with open('bot_system.log', 'a') as f:
            f.write(f"{error_msg}\n")
        
        self.add_log(error_msg)

    def set_clean(self, value: bool):
        """Enable/disable clean messages"""
        self.clean_messages = value
        self.db.clean_messages = value
        self.add_log(f"Clean messages set to: {'enabled' if value else 'disabled'}")

    def add_blacklist(self, user: str):
        """Add user to blacklist"""
        if user:
            self.db.set_setting(f"blacklist_{user}", True)
            self.add_log(f"User {user} added to blacklist")
        else:
            self.add_log("Please enter a username")

    def set_operator(self, value: str):
        """Set operator value"""
        if value:
            self.db.set_setting("operator", value)
            self.add_log(f"Operator set to: {value}")
        else:
            self.add_log("Please enter a value")

    def toggle_pause(self):
        """Toggle pause/resume state"""
        self.db.paused = not self.db.paused
        self.add_log(f"Paused: {'True' if self.db.paused else 'False'}")

    def reset_sessions(self):
        """Reset all user sessions"""
        self.db.delete_settings_like('fixed_llm_%')
        self.add_log("All user sessions reset")

    def get_db(self):
        """Get database connection"""
        return self.db

    def close(self):
        """Close UI window"""
        self.running = False
        self.root.destroy()
        self.thread.join()

    def __del__(self):
        """Cleanup on object destruction"""
        self.close()
```
