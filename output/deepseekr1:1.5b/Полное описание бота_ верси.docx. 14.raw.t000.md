class TelegramHandler:
    def __init__(self, token: str):
        self.token = token

    async def handle(self, update: Update) -> None:
        user = update.user_data['user']
        message = update.message.data.get('message', '')
        chat_id = update.message.data.get('chat_id', 0)

        if user in self.db.is_blacklisted:
            await self.ignore_message(user, message)
        
        await self.add_to_queue(item=user, message=message, chat_id=chat_id)

    """Игнорирует сообщение, если пользователь в чёрном списке."""
