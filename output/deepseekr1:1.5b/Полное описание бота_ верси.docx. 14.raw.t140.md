async def get date_since(time_str: str):
    from typing import Optional
    from dateutil.relativize import datesince
    try:
        date = datetime.datetime.strptime(time_str, '%Y-%m-%d %H:%M')
        if time_str == 'initial':
            return await datesince(date)
    except Exception as e:
        print(f"Error: {e}")
        await handle_error(error=e)
    
    last_hour = (date + datetime.timedelta(days=2) - 
                datetime.datetime.strptime(datetime.current_time(), '%Y-%m-%d %H:%M'))
    
    return datesince(last_hour)
